cmake_minimum_required(VERSION 3.24)
project(BlurCryptoSlice LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
get_filename_component(BLUR_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
set(KIT "${BLUR_ROOT}/local/recomp-kit-static-binding")
add_executable(crypto_tests "${KIT}/runtime/crypto.cpp" "${CMAKE_CURRENT_SOURCE_DIR}/crypto_tests.cpp")
target_include_directories(crypto_tests PRIVATE "${KIT}/runtime")
target_compile_definitions(crypto_tests PRIVATE GUEST_SIZE=0x10000u)
target_compile_options(crypto_tests PRIVATE -Wall -Wextra -Werror)
add_library(crypto_startup MODULE "${KIT}/runtime/crypto.cpp" "${CMAKE_CURRENT_SOURCE_DIR}/install.cpp")
target_include_directories(crypto_startup PRIVATE "${KIT}/runtime")
target_compile_definitions(crypto_startup PRIVATE GUEST_SIZE=0x785c3000u)
target_compile_options(crypto_startup PRIVATE -Wall -Wextra -Werror)
target_link_options(crypto_startup PRIVATE -Wl,-undefined,dynamic_lookup)
set_target_properties(crypto_startup PROPERTIES PREFIX "" SUFFIX ".dylib")
enable_testing()
add_test(NAME crypto_tests COMMAND crypto_tests)
