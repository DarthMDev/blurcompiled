cmake_minimum_required(VERSION 3.24)
project(BlurNative VERSION 0.0.1 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
find_package(SDL3 3.2 CONFIG REQUIRED)
find_package(ZLIB REQUIRED)
add_library(blur-assets src/assets/assets.cpp)
target_include_directories(blur-assets PUBLIC src)
target_link_libraries(blur-assets PRIVATE ZLIB::ZLIB)
add_executable(blur-native src/main.cpp)
target_link_libraries(blur-native PRIVATE SDL3::SDL3 blur-assets)
add_executable(blur-asset-probe tools/native_asset_probe.cpp)
target_link_libraries(blur-asset-probe PRIVATE blur-assets)
option(BLUR_VIDEO_SLICE "Build the native movie resource viewer using installed FFmpeg" OFF)
if(BLUR_VIDEO_SLICE)
  find_package(PkgConfig REQUIRED)
  pkg_check_modules(BLUR_FFMPEG REQUIRED IMPORTED_TARGET libavformat libavcodec libavutil libswresample)
  add_executable(blur-movie src/video/movie_main.cpp src/video/movie_audio.cpp)
  target_link_libraries(blur-movie PRIVATE SDL3::SDL3 ZLIB::ZLIB PkgConfig::BLUR_FFMPEG)
  if(CMAKE_CXX_COMPILER_ID MATCHES "Clang|GNU")
    target_compile_options(blur-movie PRIVATE -Wall -Wextra -Wpedantic)
  endif()
endif()
if(CMAKE_CXX_COMPILER_ID MATCHES "Clang|GNU")
  foreach(target blur-native blur-assets blur-asset-probe)
    target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic)
  endforeach()
endif()

option(BLUR_RECOMP_STARTUP_TESTS "Test the new startup adapters with synthetic guest memory" OFF)
if(BLUR_RECOMP_STARTUP_TESTS)
  add_executable(blur-startup-tests tests/startup_adapter_tests.cpp recomp/native/startup.cpp)
  target_include_directories(blur-startup-tests PRIVATE ${CMAKE_SOURCE_DIR}/local/recomp-kit)
  target_compile_definitions(blur-startup-tests PRIVATE GUEST_HEAP_BASE=0x02000000u)
  add_test(NAME startup-adapter COMMAND blur-startup-tests)
endif()
enable_testing()
add_test(NAME bootstrap-help COMMAND blur-native --help)
find_package(Python3 COMPONENTS Interpreter REQUIRED)
if(BLUR_VIDEO_SLICE)
  find_program(BLUR_FFMPEG_EXECUTABLE ffmpeg REQUIRED)
  add_test(NAME native-movie COMMAND ${Python3_EXECUTABLE} ${CMAKE_SOURCE_DIR}/tests/test_native_movie.py $<TARGET_FILE:blur-movie>)
endif()
add_test(NAME native-assets COMMAND ${Python3_EXECUTABLE} ${CMAKE_SOURCE_DIR}/tests/test_native_assets.py $<TARGET_FILE:blur-asset-probe>)

option(BLUR_PLUME_RENDERER "Build the native Plume/Metal renderer and validation host" OFF)
if(BLUR_PLUME_RENDERER)
  include(cmake/Plume.cmake)
endif()

# Opt-in, local-only translated slice. No generated game code enters source.
option(BLUR_RECOMP_SLICE "Build the pinned recomp-kit CRC experiment" OFF)
if(BLUR_RECOMP_SLICE)
  enable_language(C)
  set(RECOMP_KIT ${CMAKE_SOURCE_DIR}/local/recomp-kit)
  set(RECOMP_PYTHON ${CMAKE_SOURCE_DIR}/local/recomp-venv/bin/python)
  set(RECOMP_SLICE ${CMAKE_BINARY_DIR}/recomp-slice.c)
  set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS
    ${CMAKE_SOURCE_DIR}/tools/recomp_slice.py
    ${CMAKE_SOURCE_DIR}/recomp/kit-pin.json
    ${CMAKE_SOURCE_DIR}/recomp/game.toml
    ${RECOMP_KIT}/runtime/x86.h
    ${RECOMP_KIT}/tools/recomp/translate.py
    ${CMAKE_SOURCE_DIR}/local/recomp-blur/repaired/Blur.exe/functions/00cb6460.asm)
  execute_process(COMMAND ${RECOMP_PYTHON} ${CMAKE_SOURCE_DIR}/tools/recomp_slice.py
    generate --out ${RECOMP_SLICE} COMMAND_ERROR_IS_FATAL ANY)
  add_library(blur-recomp-slice SHARED tools/recomp_slice_host.c ${RECOMP_SLICE})
  target_include_directories(blur-recomp-slice PRIVATE ${RECOMP_KIT}/runtime)
  target_compile_definitions(blur-recomp-slice PRIVATE RECOMP_NULL_CHECKS=1)
  add_test(NAME recomp-crc-parity COMMAND ${RECOMP_PYTHON}
    ${CMAKE_SOURCE_DIR}/tools/recomp_slice.py verify --library $<TARGET_FILE:blur-recomp-slice>
    --out ${CMAKE_SOURCE_DIR}/reports/recomp-crc-parity.json)
endif()
